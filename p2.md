# الوحدة 6: تشفير الباسورد (Password Hashing)

**الهدف:** نخزن الباسورد بطريقة آمنة.

## المشكلة

لو خزنت `Admin@123` كنص عادي في الداتا بيز، وأي حد شاف الجدول (أو اتسرب)، يعرف كل الباسوردات.

## الحل: Hash + Salt

- **Hash:** عملية في اتجاه واحد. من الباسورد بتطلع بصمة، لكن من البصمة **متقدرش ترجع** للباسورد.
- **Salt:** نص عشوائي بنضيفه لكل مستخدم قبل الـ Hash. فلو مستخدمين اتنين باسورد واحد، البصمة بتاعتهم تطلع مختلفة. وبيصعّب هجمات الجداول الجاهزة (Rainbow Tables).
- **PBKDF2:** خوارزمية معتمدة مبنية في .NET، وبتكرر الحساب مئات الآلاف من المرات (Iterations) عشان تبطّأ أي حد بيجرّب تخمينات.
- **وقت تسجيل الدخول:** بناخد الباسورد اللي كتبه المستخدم، نعمله Hash بنفس الـ Salt، ونقارن بالمخزن.

**Security/PasswordHasher.cs**

```csharp
using System;
using System.Security.Cryptography;

namespace InventorySystem.Security
{
    public static class PasswordHasher
    {
        private const int SaltSize = 16;       // 128 bit
        private const int HashSize = 32;       // 256 bit
        private const int Iterations = 600000; // عدد التكرارات

        // الشكل المخزن: iterations.salt.hash  (Base64)
        public static string Hash(string password)
        {
            byte[] salt = new byte[SaltSize];
            using (var rng = RandomNumberGenerator.Create())
            {
                rng.GetBytes(salt);
            }

            byte[] hash;
            using (var pbkdf2 = new Rfc2898DeriveBytes(password, salt, Iterations, HashAlgorithmName.SHA256))
            {
                hash = pbkdf2.GetBytes(HashSize);
            }

            return Iterations + "." + Convert.ToBase64String(salt) + "." + Convert.ToBase64String(hash);
        }

        public static bool Verify(string password, string storedHash)
        {
            if (string.IsNullOrEmpty(password) || string.IsNullOrEmpty(storedHash))
                return false;

            string[] parts = storedHash.Split('.');
            if (parts.Length != 3) return false;

            int iterations;
            if (!int.TryParse(parts[0], out iterations)) return false;

            byte[] salt = Convert.FromBase64String(parts[1]);
            byte[] expected = Convert.FromBase64String(parts[2]);

            byte[] actual;
            using (var pbkdf2 = new Rfc2898DeriveBytes(password, salt, iterations, HashAlgorithmName.SHA256))
            {
                actual = pbkdf2.GetBytes(expected.Length);
            }

            return FixedTimeEquals(actual, expected);
        }

        // مقارنة بتاخد نفس الوقت دايمًا (بتمنع Timing Attacks)
        private static bool FixedTimeEquals(byte[] a, byte[] b)
        {
            if (a.Length != b.Length) return false;
            int diff = 0;
            for (int i = 0; i < a.Length; i++)
                diff |= a[i] ^ b[i];
            return diff == 0;
        }
    }
}
```

> **للمناقشة:** "بنستخدم PBKDF2 مع Salt عشوائي لكل مستخدم وعدد تكرارات عالي، وبنقارن بطريقة Constant-time. ومبنخزنش الباسورد الأصلي أبدًا."

## 7.2 UserRepository

فيه 3 وظايف: تسجيل الدخول (`Authenticate`)، إنشاء مستخدم (`Create`)، وإنشاء المستخدمين الافتراضيين أول مرة (`SeedDefaultUsers`).

**Data/UserRepository.cs**

```csharp
using System;
using System.Data.SqlClient;
using InventorySystem.Models;
using InventorySystem.Security;

namespace InventorySystem.Data
{
    public class UserRepository
    {
        // Hash وهمي: بنستخدمه لو اسم المستخدم مش موجود،
        // عشان وقت الرد يبقى نفس الوقت (بيصعّب تخمين أسماء المستخدمين)
        private static readonly string DummyHash = PasswordHasher.Hash("dummy-password");

        public User Authenticate(string username, string password)
        {
            User user = null;
            string storedHash = null;

            using (var con = Db.GetConnection())
            using (var cmd = new SqlCommand(
                "SELECT UserId, Username, PasswordHash, Role FROM Users WHERE Username = @u AND IsActive = 1", con))
            {
                cmd.Parameters.Add("@u", System.Data.SqlDbType.NVarChar, 50).Value = username;
                con.Open();
                using (var r = cmd.ExecuteReader())
                {
                    if (r.Read())
                    {
                        user = new User
                        {
                            UserId = r.GetInt32(0),
                            Username = r.GetString(1),
                            Role = (Role)Enum.Parse(typeof(Role), r.GetString(3))
                        };
                        storedHash = r.GetString(2);
                    }
                }
            }

            if (user == null)
            {
                PasswordHasher.Verify(password, DummyHash);
                return null;
            }

            return PasswordHasher.Verify(password, storedHash) ? user : null;
        }

        public void Create(string username, string password, Role role)
        {
            using (var con = Db.GetConnection())
            using (var cmd = new SqlCommand(
                "INSERT INTO Users (Username, PasswordHash, Role) VALUES (@u, @h, @r)", con))
            {
                cmd.Parameters.Add("@u", System.Data.SqlDbType.NVarChar, 50).Value = username;
                cmd.Parameters.Add("@h", System.Data.SqlDbType.NVarChar, 255).Value = PasswordHasher.Hash(password);
                cmd.Parameters.Add("@r", System.Data.SqlDbType.NVarChar, 20).Value = role.ToString();
                con.Open();
                cmd.ExecuteNonQuery();
            }
        }

        // لو مفيش ولا مستخدم، بنعمل admin و viewer
        public void SeedDefaultUsers()
        {
            int count;
            using (var con = Db.GetConnection())
            using (var cmd = new SqlCommand("SELECT COUNT(*) FROM Users", con))
            {
                con.Open();
                count = (int)cmd.ExecuteScalar();
            }

            if (count > 0) return;

            Create("admin", "Admin@123", Role.Admin);
            Create("viewer", "Viewer@123", Role.Viewer);
        }
    }
}
```

**الحسابات اللي هتتعمل أوتوماتيك:**

| Username | Password | Role |
|---|---|---|
| admin | Admin@123 | Admin |
| viewer | Viewer@123 | Viewer |

## 7.3 AuditRepository

**الفكرة:** أي إضافة/تعديل/مسح بيتسجل. والتسجيل بيحصل **في نفس الـ Transaction** بتاعة العملية. يعني لو التسجيل فشل، العملية كلها بتتلغي، ومفيش تعديل يحصل من غير أثر.

**Data/AuditRepository.cs**

```csharp
using System.Data;
using System.Data.SqlClient;
using InventorySystem.Security;

namespace InventorySystem.Data
{
    public class AuditRepository
    {
        // بتتنادى من جوه ProductRepository بنفس الـ Connection والـ Transaction
        public static void Write(SqlConnection con, SqlTransaction tx, string actionType,
                                 int? productId, string productName, string oldValues, string newValues)
        {
            const string sql = @"
INSERT INTO AuditLogs (ActionType, ProductId, ProductName, UserId, Username, OldValues, NewValues)
VALUES (@a, @pid, @pn, @uid, @un, @old, @new)";

            using (var cmd = new SqlCommand(sql, con, tx))
            {
                cmd.Parameters.Add("@a", SqlDbType.NVarChar, 10).Value = actionType;
                cmd.Parameters.Add("@pid", SqlDbType.Int).Value = Db.ToDb(productId);
                cmd.Parameters.Add("@pn", SqlDbType.NVarChar, 100).Value = Db.ToDb(productName);
                cmd.Parameters.Add("@uid", SqlDbType.Int).Value = Session.CurrentUser.UserId;
                cmd.Parameters.Add("@un", SqlDbType.NVarChar, 50).Value = Session.CurrentUser.Username;
                cmd.Parameters.Add("@old", SqlDbType.NVarChar, -1).Value = Db.ToDb(oldValues);
                cmd.Parameters.Add("@new", SqlDbType.NVarChar, -1).Value = Db.ToDb(newValues);
                cmd.ExecuteNonQuery();
            }
        }

        public DataTable GetAll()
        {
            var dt = new DataTable("AuditLogs");
            const string sql = @"
SELECT TOP 500 AuditId, ChangedAt, ActionType, ProductId, ProductName, Username, OldValues, NewValues
FROM AuditLogs
ORDER BY ChangedAt DESC, AuditId DESC";

            using (var con = Db.GetConnection())
            using (var da = new SqlDataAdapter(sql, con))
            {
                da.Fill(dt);
            }
            return dt;
        }
    }
}
```

> `NVarChar, -1` معناها `NVARCHAR(MAX)`.

## 7.4 ProductRepository (أهم كلاس)

اللي بيعمله:

- `Search(...)`: بحث وفلترة (بالاسم / الفئة / حالة المخزون).
- `GetReportTable(...)`: الداتا بتاعة التقرير.
- `GetById`, `Add`, `Update`, `Delete`: وكلهم بيتأكدوا إن المستخدم **Admin**، وبيسجلوا Audit.

**حالات المخزون (Stock Status):**

| الحالة | الشرط |
|---|---|
| Out of Stock | الكمية = 0 |
| Low Stock | الكمية أكبر من 0 وأقل من الـ Threshold (10 افتراضيًا) |
| In Stock | الكمية = الـ Threshold أو أكتر |

**Data/ProductRepository.cs**

```csharp
using System;
using System.Data;
using System.Data.SqlClient;
using System.Text;
using InventorySystem.Models;
using InventorySystem.Security;

namespace InventorySystem.Data
{
    public class ProductRepository
    {
        // Query أساسي مشترك بين البحث والتقرير. الأعمدة هنا لازم تتطابق مع الـ XSD في الوحدة 12.
        private const string BaseSelect = @"
SELECT * FROM (
    SELECT  p.ProductId,
            p.Name,
            p.Description,
            p.Quantity,
            p.Price,
            p.SupplierName,
            p.CategoryId,
            ISNULL(c.Name, 'Uncategorized') AS CategoryName,
            CAST(p.Quantity * p.Price AS DECIMAL(18,2)) AS TotalValue,
            CASE WHEN p.Quantity = 0           THEN 'Out of Stock'
                 WHEN p.Quantity < @Threshold  THEN 'Low Stock'
                 ELSE 'In Stock' END AS StockStatus,
            CAST(CASE WHEN p.Quantity < @Threshold THEN 1 ELSE 0 END AS BIT) AS IsLowStock
    FROM Products p
    LEFT JOIN Categories c ON c.CategoryId = p.CategoryId
) AS X";

        // ---------- البحث والفلترة ----------
        public DataTable Search(string text, int? categoryId, string stockStatus, int threshold)
        {
            var sql = new StringBuilder(BaseSelect);
            sql.Append(" WHERE 1 = 1");

            bool hasText = !string.IsNullOrWhiteSpace(text);
            if (hasText)
                sql.Append(" AND (Name LIKE @text OR SupplierName LIKE @text OR Description LIKE @text)");
            if (categoryId.HasValue)
                sql.Append(" AND CategoryId = @cat");
            if (!string.IsNullOrEmpty(stockStatus))
                sql.Append(" AND StockStatus = @status");

            sql.Append(" ORDER BY Name");

            var dt = new DataTable("Products");
            using (var con = Db.GetConnection())
            using (var cmd = new SqlCommand(sql.ToString(), con))
            using (var da = new SqlDataAdapter(cmd))
            {
                cmd.Parameters.Add("@Threshold", SqlDbType.Int).Value = threshold;
                if (hasText)
                    cmd.Parameters.Add("@text", SqlDbType.NVarChar, 200).Value = "%" + text.Trim() + "%";
                if (categoryId.HasValue)
                    cmd.Parameters.Add("@cat", SqlDbType.Int).Value = categoryId.Value;
                if (!string.IsNullOrEmpty(stockStatus))
                    cmd.Parameters.Add("@status", SqlDbType.NVarChar, 20).Value = stockStatus;

                da.Fill(dt);   // Fill بيفتح ويقفل الـ Connection لوحده
            }
            return dt;
        }

        // ---------- داتا التقرير ----------
        public DataTable GetReportTable(int threshold, bool lowStockOnly)
        {
            string sql = BaseSelect
                + (lowStockOnly ? " WHERE IsLowStock = 1" : "")
                + " ORDER BY IsLowStock DESC, Name";

            var dt = new DataTable("Products");
            using (var con = Db.GetConnection())
            using (var cmd = new SqlCommand(sql, con))
            using (var da = new SqlDataAdapter(cmd))
            {
                cmd.Parameters.Add("@Threshold", SqlDbType.Int).Value = threshold;
                da.Fill(dt);
            }
            return dt;
        }

        // ---------- جلب منتج واحد ----------
        public Product GetById(int id)
        {
            using (var con = Db.GetConnection())
            {
                con.Open();
                return GetById(con, null, id);
            }
        }

        private static Product GetById(SqlConnection con, SqlTransaction tx, int id)
        {
            const string sql = @"SELECT ProductId, Name, Description, Quantity, Price, SupplierName, CategoryId
                                 FROM Products WHERE ProductId = @id";
            using (var cmd = new SqlCommand(sql, con, tx))
            {
                cmd.Parameters.Add("@id", SqlDbType.Int).Value = id;
                using (var r = cmd.ExecuteReader())
                {
                    if (!r.Read()) return null;
                    return new Product
                    {
                        ProductId = r.GetInt32(0),
                        Name = r.GetString(1),
                        Description = r.IsDBNull(2) ? null : r.GetString(2),
                        Quantity = r.GetInt32(3),
                        Price = r.GetDecimal(4),
                        SupplierName = r.GetString(5),
                        CategoryId = r.IsDBNull(6) ? (int?)null : r.GetInt32(6)
                    };
                }
            }
        }

        // ---------- إضافة ----------
        public int Add(Product p)
        {
            Session.RequireAdmin();

            using (var con = Db.GetConnection())
            {
                con.Open();
                using (var tx = con.BeginTransaction())
                {
                    try
                    {
                        const string sql = @"
INSERT INTO Products (Name, Description, Quantity, Price, SupplierName, CategoryId)
OUTPUT INSERTED.ProductId
VALUES (@name, @desc, @qty, @price, @supplier, @cat)";

                        int newId;
                        using (var cmd = new SqlCommand(sql, con, tx))
                        {
                            AddParams(cmd, p);
                            newId = (int)cmd.ExecuteScalar();
                        }

                        p.ProductId = newId;
                        AuditRepository.Write(con, tx, "INSERT", newId, p.Name, null, p.ToAuditString());

                        tx.Commit();
                        return newId;
                    }
                    catch
                    {
                        tx.Rollback();
                        throw;
                    }
                }
            }
        }

        // ---------- تعديل ----------
        public void Update(Product p)
        {
            Session.RequireAdmin();

            using (var con = Db.GetConnection())
            {
                con.Open();
                using (var tx = con.BeginTransaction())
                {
                    try
                    {
                        Product old = GetById(con, tx, p.ProductId);
                        if (old == null) throw new InvalidOperationException("Product no longer exists.");

                        const string sql = @"
UPDATE Products
SET Name = @name, Description = @desc, Quantity = @qty, Price = @price,
    SupplierName = @supplier, CategoryId = @cat
WHERE ProductId = @id";

                        using (var cmd = new SqlCommand(sql, con, tx))
                        {
                            AddParams(cmd, p);
                            cmd.Parameters.Add("@id", SqlDbType.Int).Value = p.ProductId;
                            cmd.ExecuteNonQuery();
                        }

                        AuditRepository.Write(con, tx, "UPDATE", p.ProductId, p.Name,
                                              old.ToAuditString(), p.ToAuditString());
                        tx.Commit();
                    }
                    catch
                    {
                        tx.Rollback();
                        throw;
                    }
                }
            }
        }

        // ---------- حذف ----------
        public void Delete(int id)
        {
            Session.RequireAdmin();

            using (var con = Db.GetConnection())
            {
                con.Open();
                using (var tx = con.BeginTransaction())
                {
                    try
                    {
                        Product old = GetById(con, tx, id);
                        if (old == null) throw new InvalidOperationException("Product no longer exists.");

                        using (var cmd = new SqlCommand("DELETE FROM Products WHERE ProductId = @id", con, tx))
                        {
                            cmd.Parameters.Add("@id", SqlDbType.Int).Value = id;
                            cmd.ExecuteNonQuery();
                        }

                        AuditRepository.Write(con, tx, "DELETE", id, old.Name, old.ToAuditString(), null);
                        tx.Commit();
                    }
                    catch
                    {
                        tx.Rollback();
                        throw;
                    }
                }
            }
        }

        private static void AddParams(SqlCommand cmd, Product p)
        {
            cmd.Parameters.Add("@name", SqlDbType.NVarChar, 100).Value = p.Name;
            cmd.Parameters.Add("@desc", SqlDbType.NVarChar, 500).Value = Db.ToDb(p.Description);
            cmd.Parameters.Add("@qty", SqlDbType.Int).Value = p.Quantity;
            cmd.Parameters.Add("@price", SqlDbType.Decimal).Value = p.Price;
            cmd.Parameters["@price"].Precision = 18;
            cmd.Parameters["@price"].Scale = 2;
            cmd.Parameters.Add("@supplier", SqlDbType.NVarChar, 100).Value = p.SupplierName;
            cmd.Parameters.Add("@cat", SqlDbType.Int).Value = Db.ToDb(p.CategoryId);
        }
    }
}
```

## شرح الأجزاء الصعبة

1. **ليه `SELECT * FROM (...) AS X`؟** عشان نقدر نعمل `WHERE` على `StockStatus` اللي هو عمود محسوب (Computed). مينفعش تستخدم alias في نفس مستوى الـ WHERE.
2. **ليه Transaction؟** لو `INSERT` نجح و`AuditLogs` فشل، الـ `Rollback` بيلغي الاتنين.
3. **ليه `Session.RequireAdmin()` هنا مع إن الزرار متخفي للـ Viewer؟** لأن الحماية الحقيقية لازم تبقى في الـ Logic مش في الواجهة بس. دي اسمها **Defense in Depth**.
4. **`OUTPUT INSERTED.ProductId`:** بيرجّع الـ ID اللي الداتا بيز ولّدته بدل استعلام تاني.

## 8.2 كود الشاشة

افتح `LoginForm.cs` (كليك يمين ← View Code) واستبدل الكلاس كله بالكود ده:

```csharp
using System;
using System.Data.SqlClient;
using System.Windows.Forms;
using InventorySystem.Data;
using InventorySystem.Security;

namespace InventorySystem.Forms
{
    public partial class LoginForm : Form
    {
        private readonly UserRepository _users = new UserRepository();
        private int _failedAttempts;

        public LoginForm()
        {
            InitializeComponent();
            btnLogin.Click += btnLogin_Click;
        }

        private void btnLogin_Click(object sender, EventArgs e)
        {
            lblError.Text = "";
            string username = txtUsername.Text.Trim();
            string password = txtPassword.Text;

            if (username.Length == 0 || password.Length == 0)
            {
                lblError.Text = "Please enter username and password.";
                return;
            }

            try
            {
                var user = _users.Authenticate(username, password);

                if (user == null)
                {
                    _failedAttempts++;
                    // رسالة عامة. متقولش "الباسورد غلط" أو "المستخدم مش موجود"
                    lblError.Text = "Invalid username or password.";
                    txtPassword.Clear();
                    txtPassword.Focus();

                    if (_failedAttempts >= 5)
                    {
                        MessageBox.Show("Too many failed attempts. The application will close.",
                            "Login", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                        DialogResult = DialogResult.Cancel;
                    }
                    return;
                }

                Session.CurrentUser = user;
                DialogResult = DialogResult.OK;   // بيقفل الشاشة وبيرجّع OK
            }
            catch (SqlException ex)
            {
                MessageBox.Show("Database error:\n" + ex.Message, "Error",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }
    }
}
```

## 8.3 Program.cs

افتح `Program.cs` واستبدل محتواه بالكامل:

```csharp
using System;
using System.Data.SqlClient;
using System.Windows.Forms;
using InventorySystem.Data;
using InventorySystem.Forms;
using InventorySystem.Security;

namespace InventorySystem
{
    internal static class Program
    {
        [STAThread]
        static void Main()
        {
            Application.EnableVisualStyles();
            Application.SetCompatibleTextRenderingDefault(false);

            try
            {
                new UserRepository().SeedDefaultUsers();   // أول مرة بس
            }
            catch (SqlException ex)
            {
                MessageBox.Show("Cannot connect to the database.\n\n" + ex.Message,
                    "Startup error", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            // دورة: Login -> Main -> (Logout -> Login تاني)
            while (true)
            {
                using (var login = new LoginForm())
                {
                    if (login.ShowDialog() != DialogResult.OK) break;
                }

                bool loggedOut;
                using (var main = new MainForm())
                {
                    main.ShowDialog();
                    loggedOut = main.LoggedOut;
                }

                Session.CurrentUser = null;
                if (!loggedOut) break;   // قفل الشاشة بالـ X = خروج من البرنامج
            }
        }
    }
}
```
