## 9.2 الكود

افتح `MainForm.cs` ← View Code ← استبدل كل الكلاس:

```csharp
using System;
using System.Collections.Generic;
using System.Data;
using System.Drawing;
using System.Windows.Forms;
using InventorySystem.Data;
using InventorySystem.Models;
using InventorySystem.Security;

namespace InventorySystem.Forms
{
    public partial class MainForm : Form
    {
        private readonly ProductRepository _products = new ProductRepository();
        private readonly CategoryRepository _categories = new CategoryRepository();

        // Program.cs بيقراها عشان يعرف: هل المستخدم عمل Logout ولا قفل البرنامج؟
        public bool LoggedOut { get; private set; }

        public MainForm()
        {
            InitializeComponent();

            Load += MainForm_Load;
            btnSearch.Click += delegate { LoadProducts(); };
            btnClear.Click += btnClear_Click;
            btnAdd.Click += btnAdd_Click;
            btnEdit.Click += delegate { EditSelected(); };
            btnDelete.Click += btnDelete_Click;
            btnReport.Click += btnReport_Click;
            btnAudit.Click += btnAudit_Click;
            btnLogout.Click += btnLogout_Click;

            cboCategory.SelectionChangeCommitted += delegate { LoadProducts(); };
            cboStatus.SelectionChangeCommitted += delegate { LoadProducts(); };

            dgvProducts.DataBindingComplete += dgvProducts_DataBindingComplete;
            dgvProducts.CellDoubleClick += (s, e) =>
            {
                if (Session.IsAdmin && e.RowIndex >= 0) EditSelected();
            };

            AcceptButton = btnSearch;   // Enter في خانة البحث = بحث
        }

        private void MainForm_Load(object sender, EventArgs e)
        {
            lblUserInfo.Text = "User: " + Session.CurrentUser.Username +
                               "  |  Role: " + Session.CurrentUser.Role;

            ApplyPermissions();
            LoadFilters();
            LoadProducts();
        }

        // ---------- الصلاحيات ----------
        private void ApplyPermissions()
        {
            bool isAdmin = Session.IsAdmin;
            btnAdd.Visible = isAdmin;
            btnEdit.Visible = isAdmin;
            btnDelete.Visible = isAdmin;
            btnAudit.Visible = isAdmin;
            // Viewer يشوف: المنتجات + البحث + التقارير بس
        }

        // ---------- تجهيز الفلاتر ----------
        private void LoadFilters()
        {
            List<Category> cats = _categories.GetAll();
            cats.Insert(0, new Category { CategoryId = 0, Name = "All Categories" });
            cboCategory.DisplayMember = "Name";
            cboCategory.ValueMember = "CategoryId";
            cboCategory.DataSource = cats;

            cboStatus.Items.Clear();
            cboStatus.Items.AddRange(new object[] { "All Statuses", "In Stock", "Low Stock", "Out of Stock" });
            cboStatus.SelectedIndex = 0;
        }

        // ---------- تحميل المنتجات (بحث + فلترة) ----------
        private void LoadProducts()
        {
            try
            {
                string text = txtSearch.Text.Trim();

                int? categoryId = null;
                int catValue = Convert.ToInt32(cboCategory.SelectedValue ?? 0);
                if (catValue > 0) categoryId = catValue;

                string status = cboStatus.SelectedIndex > 0 ? cboStatus.Text : null;

                dgvProducts.DataSource = _products.Search(text, categoryId, status, AppSettings.LowStockThreshold);
            }
            catch (Exception ex)
            {
                MessageBox.Show("Could not load products:\n" + ex.Message, "Error",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        // بيتنفذ بعد ما الجدول يتملي: عناوين الأعمدة + تلوين الصفوف
        private void dgvProducts_DataBindingComplete(object sender, DataGridViewBindingCompleteEventArgs e)
        {
            if (dgvProducts.Columns.Count == 0) return;

            SetHeader("ProductId", "ID");
            SetHeader("Name", "Product Name");
            SetHeader("Description", "Description");
            SetHeader("Quantity", "Qty in Stock");
            SetHeader("Price", "Price");
            SetHeader("SupplierName", "Supplier");
            SetHeader("CategoryName", "Category");
            SetHeader("TotalValue", "Total Value");
            SetHeader("StockStatus", "Stock Status");

            HideColumn("CategoryId");
            HideColumn("IsLowStock");

            if (dgvProducts.Columns["Price"] != null)
                dgvProducts.Columns["Price"].DefaultCellStyle.Format = "N2";
            if (dgvProducts.Columns["TotalValue"] != null)
                dgvProducts.Columns["TotalValue"].DefaultCellStyle.Format = "N2";

            foreach (DataGridViewRow row in dgvProducts.Rows)
            {
                string status = Convert.ToString(row.Cells["StockStatus"].Value);
                if (status == "Out of Stock")
                    row.DefaultCellStyle.BackColor = Color.MistyRose;
                else if (status == "Low Stock")
                    row.DefaultCellStyle.BackColor = Color.LemonChiffon;
            }
        }

        private void SetHeader(string column, string header)
        {
            if (dgvProducts.Columns[column] != null)
                dgvProducts.Columns[column].HeaderText = header;
        }

        private void HideColumn(string column)
        {
            if (dgvProducts.Columns[column] != null)
                dgvProducts.Columns[column].Visible = false;
        }

        private int? GetSelectedProductId()
        {
            if (dgvProducts.CurrentRow == null) return null;
            return Convert.ToInt32(dgvProducts.CurrentRow.Cells["ProductId"].Value);
        }

        // ---------- الأزرار ----------
        private void btnClear_Click(object sender, EventArgs e)
        {
            txtSearch.Clear();
            cboCategory.SelectedIndex = 0;
            cboStatus.SelectedIndex = 0;
            LoadProducts();
        }

        private void btnAdd_Click(object sender, EventArgs e)
        {
            using (var f = new ProductForm())
            {
                if (f.ShowDialog(this) == DialogResult.OK) LoadProducts();
            }
        }

        private void EditSelected()
        {
            int? id = GetSelectedProductId();
            if (id == null)
            {
                MessageBox.Show("Select a product first.");
                return;
            }

            using (var f = new ProductForm(id.Value))
            {
                if (f.ShowDialog(this) == DialogResult.OK) LoadProducts();
            }
        }

        private void btnDelete_Click(object sender, EventArgs e)
        {
            int? id = GetSelectedProductId();
            if (id == null)
            {
                MessageBox.Show("Select a product first.");
                return;
            }

            string name = Convert.ToString(dgvProducts.CurrentRow.Cells["Name"].Value);
            var answer = MessageBox.Show("Delete product \"" + name + "\"?\nThis cannot be undone.",
                "Confirm delete", MessageBoxButtons.YesNo, MessageBoxIcon.Warning);
            if (answer != DialogResult.Yes) return;

            try
            {
                _products.Delete(id.Value);
                LoadProducts();
            }
            catch (UnauthorizedAccessException ex)
            {
                MessageBox.Show(ex.Message, "Access denied", MessageBoxButtons.OK, MessageBoxIcon.Stop);
            }
            catch (Exception ex)
            {
                MessageBox.Show("Delete failed:\n" + ex.Message, "Error",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnReport_Click(object sender, EventArgs e)
        {
            using (var f = new ReportForm())
            {
                f.ShowDialog(this);
            }
        }

        private void btnAudit_Click(object sender, EventArgs e)
        {
            using (var f = new AuditLogForm())
            {
                f.ShowDialog(this);
            }
        }

        private void btnLogout_Click(object sender, EventArgs e)
        {
            LoggedOut = true;
            Close();
        }
    }
}
```

## 10.2 الكود

```csharp
using System;
using System.Windows.Forms;
using InventorySystem.Data;
using InventorySystem.Models;

namespace InventorySystem.Forms
{
    public partial class ProductForm : Form
    {
        private readonly ProductRepository _repo = new ProductRepository();
        private readonly CategoryRepository _catRepo = new CategoryRepository();
        private readonly int? _productId;   // null = إضافة ، له قيمة = تعديل

        public ProductForm() : this(null) { }

        public ProductForm(int? productId)
        {
            InitializeComponent();
            _productId = productId;
            Load += ProductForm_Load;
            btnSave.Click += btnSave_Click;
        }

        private void ProductForm_Load(object sender, EventArgs e)
        {
            var cats = _catRepo.GetAll();
            cats.Insert(0, new Category { CategoryId = 0, Name = "(None)" });
            cboCategory.DisplayMember = "Name";
            cboCategory.ValueMember = "CategoryId";
            cboCategory.DataSource = cats;

            if (!_productId.HasValue)
            {
                Text = "Add Product";
                lblIdValue.Text = "(auto)";
                return;
            }

            Text = "Edit Product";
            Product p = _repo.GetById(_productId.Value);
            if (p == null)
            {
                MessageBox.Show("This product no longer exists.");
                DialogResult = DialogResult.Cancel;
                BeginInvoke((Action)Close);
                return;
            }

            lblIdValue.Text = p.ProductId.ToString();
            txtName.Text = p.Name;
            txtDescription.Text = p.Description;
            nudQuantity.Value = p.Quantity;
            nudPrice.Value = p.Price;
            txtSupplier.Text = p.SupplierName;
            cboCategory.SelectedValue = p.CategoryId ?? 0;
        }

        private void btnSave_Click(object sender, EventArgs e)
        {
            // ---------- Validation ----------
            string name = txtName.Text.Trim();
            string supplier = txtSupplier.Text.Trim();

            if (name.Length == 0)
            {
                MessageBox.Show("Product name is required.", "Validation");
                txtName.Focus();
                return;
            }
            if (supplier.Length == 0)
            {
                MessageBox.Show("Supplier name is required.", "Validation");
                txtSupplier.Focus();
                return;
            }

            int catId = Convert.ToInt32(cboCategory.SelectedValue ?? 0);

            var p = new Product
            {
                ProductId = _productId ?? 0,
                Name = name,
                Description = string.IsNullOrWhiteSpace(txtDescription.Text) ? null : txtDescription.Text.Trim(),
                Quantity = (int)nudQuantity.Value,
                Price = nudPrice.Value,
                SupplierName = supplier,
                CategoryId = catId > 0 ? (int?)catId : null
            };

            // ---------- Save ----------
            try
            {
                if (_productId.HasValue) _repo.Update(p);
                else _repo.Add(p);

                DialogResult = DialogResult.OK;
            }
            catch (UnauthorizedAccessException ex)
            {
                MessageBox.Show(ex.Message, "Access denied", MessageBoxButtons.OK, MessageBoxIcon.Stop);
            }
            catch (Exception ex)
            {
                MessageBox.Show("Save failed:\n" + ex.Message, "Error",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }
    }
}
```

# الوحدة 12: Crystal Reports

**الهدف:** تقرير بيلخّص المخزون ويظلّل المنتجات اللي قربت تخلص.

## الفكرة (Push Model)

بدل ما Crystal يتصل بالداتا بيز بنفسه (وده بيسبب مشاكل Logon كتير)، إحنا **بنجيب الداتا بالـ ADO.NET بتاعنا ونديها للتقرير** بـ `SetDataSource`. ده أبسط وأضمن.

عشان التقرير يعرف أشكال الأعمدة وقت التصميم، بنعمل **DataSet Schema (.xsd)** فاضي فيه نفس الأعمدة.

## 12.1 إنشاء الـ XSD

1. كليك يمين على فولدر **Reports** ← Add ← New Item ← اكتب في البحث `DataSet` ← اختار **DataSet** ← الاسم `InventoryDataSet.xsd` ← Add.
2. هتفتح شاشة فاضية. من الـ **Toolbox** اسحب **DataTable** على الشاشة.
3. غيّر اسم الـ DataTable لـ **`Products`** (بالظبط، بنفس الحروف).
4. كليك يمين على الـ DataTable ← Add ← **Column** وكرر لكل عمود. بعد ما تضيف كل عمود، غيّر **Name** و**DataType** من نافذة Properties:

| Column Name | DataType |
|---|---|
| ProductId | System.Int32 |
| Name | System.String |
| Description | System.String |
| Quantity | System.Int32 |
| Price | System.Decimal |
| SupplierName | System.String |
| CategoryId | System.Int32 |
| CategoryName | System.String |
| TotalValue | System.Decimal |
| StockStatus | System.String |
| IsLowStock | System.Boolean |

> ⚠️ **الأسماء والأنواع لازم تتطابق بالظبط** مع أعمدة الـ SQL في `ProductRepository.BaseSelect`. أي اختلاف = Error وقت التشغيل.

5. احفظ (Ctrl+S) واعمل Build.

## 12.2 إنشاء التقرير

1. كليك يمين على **Reports** ← Add ← New Item ← **Crystal Reports** ← الاسم `InventoryReport.rpt` ← Add.
2. هتظهر نافذة اختيار. اختار **Using the Report Wizard** ← **Standard** ← OK.
3. **Data:** وسّع **Project Data** ← **ADO.NET DataSets** ← `InventoryDataSet` ← اختار **Products** واضغط **>** (أو دبل كليك) ← Next.
4. **Fields:** ضيف الحقول دي بالترتيب: `ProductId` ، `Name` ، `CategoryName` ، `SupplierName` ، `Quantity` ، `Price` ، `TotalValue` ، `StockStatus` ← بعدها اضغط **Finish** (سيب باقي الخطوات من غير ما تعدّل).

دلوقتي عندك تقرير بالأقسام (Sections): Report Header / Page Header / Details / Report Footer / Page Footer.

## 12.3 إضافة الـ Parameter (عشان نعرض قيمة الـ Threshold)

1. افتح **Field Explorer**: من قايمة **Crystal Reports** ← Field Explorer (أو View ← Other Windows ← Field Explorer).
2. كليك يمين على **Parameter Fields** ← New:
   - Name: `Threshold`
   - Type: **Number**
   - اضغط OK.
3. اسحب `?Threshold` من Field Explorer وحطه في **Report Header**. وجنبه Text Object (Insert ← Text Object) مكتوب فيه: `Low stock threshold:`.

## 12.4 تحسين شكل التقرير

- **عنوان التقرير:** في Report Header: Insert ← Text Object ← اكتب `Inventory Summary Report` ← كبّر الخط وخليه Bold.
- **تاريخ الطباعة:** من Field Explorer ← Special Fields ← **Print Date** ← اسحبه في Page Header.
- **رقم الصفحة:** Special Fields ← **Page N of M** ← اسحبه في Page Footer.
- **عناوين الأعمدة:** في Page Header بتلاقي عناوين تلقائية. عدّلها لو حابب (دبل كليك).

## 12.5 تظليل المنتجات Low Stock ⭐

1. كليك يمين على المساحة الفاضية جنب **Details** ← **Section Expert**.
2. اختار **Details** من القايمة ← تاب **Color**.
3. علّم على **Background Color**، واضغط الزرار **x-2** اللي جنبها (Conditional formula).
4. الصق الفورمولا دي:

```
If {Products.IsLowStock} = True Then RGB(255, 220, 220) Else crNoColor
```

5. اضغط **Save and close** ← OK.

(اختياري) خلّي الكمية نفسها حمرا: كليك يمين على حقل `Quantity` في Details ← Format Field ← Font ← Color ← x-2 ← الصق:

```
If {Products.IsLowStock} = True Then crRed Else crBlack
```

## 12.6 الملخص في الآخر (Summary)

1. **عدد المنتجات:** Insert ← Summary ← Field = `{Products.ProductId}` ← Calculate = **Count** ← Summary location = **Grand Total (Report Footer)** ← OK.
2. **إجمالي قيمة المخزون:** Insert ← Summary ← Field = `{Products.TotalValue}` ← Calculate = **Sum** ← Grand Total ← OK.
3. **عدد المنتجات Low Stock:** Field Explorer ← كليك يمين على **Running Total Fields** ← New:
   - Name: `LowStockCount`
   - Field to summarize: `Products.ProductId`
   - Type of summary: **count**
   - Evaluate: **Use a formula** ← اضغط x-2 ← اكتب `{Products.IsLowStock} = True` ← Save
   - Reset: **Never**
   - OK، وبعدين اسحبه في **Report Footer**.
4. ضيف Text Objects جنب كل رقم: `Total products:` ، `Total inventory value:` ، `Low-stock items:`.

## 12.7 شاشة عرض التقرير (ReportForm)

كليك يمين على **Forms** ← Add ← Form ← `ReportForm.cs`.

**خصائص الـ Form:** Text = "Inventory Report" ، StartPosition = CenterParent ، WindowState = Maximized.

**الـ Controls:**

1. **Panel** ← (Name) `topPanel` ← Dock = Top ← Height = 45. وجواه:
   - **Label** Text = "Low stock below:"
   - **NumericUpDown** ← (Name) `nudThreshold` ← Minimum = 1 ← Maximum = 100000 ← Value = 10
   - **CheckBox** ← (Name) `chkLowOnly` ← Text = "Low stock items only"
   - **Button** ← (Name) `btnRefresh` ← Text = "Refresh"
2. من الـ Toolbox، تاب **Crystal Reports**، اسحب **CrystalReportViewer** ← (Name) = `crystalReportViewer1` ← **Dock = Fill** (لو غطى على البانل: كليك يمين ← Bring to Front).

**الكود:**

```csharp
using System;
using System.Data;
using System.Windows.Forms;
using CrystalDecisions.CrystalReports.Engine;
using InventorySystem.Data;
using InventorySystem.Reports;

namespace InventorySystem.Forms
{
    public partial class ReportForm : Form
    {
        private readonly ProductRepository _products = new ProductRepository();
        private ReportDocument _report;

        public ReportForm()
        {
            InitializeComponent();
            Load += ReportForm_Load;
            btnRefresh.Click += delegate { ShowReport(); };
        }

        private void ReportForm_Load(object sender, EventArgs e)
        {
            nudThreshold.Value = AppSettings.LowStockThreshold;
            crystalReportViewer1.ToolPanelView = CrystalDecisions.Windows.Forms.ToolPanelViewType.None;
            ShowReport();
        }

        private void ShowReport()
        {
            try
            {
                int threshold = (int)nudThreshold.Value;
                DataTable data = _products.GetReportTable(threshold, chkLowOnly.Checked);

                var report = new InventoryReport();
                report.SetDataSource(data);
                report.SetParameterValue("Threshold", threshold);

                crystalReportViewer1.ReportSource = report;

                if (_report != null) _report.Dispose();
                _report = report;
            }
            catch (Exception ex)
            {
                MessageBox.Show("Could not show the report:\n" + ex.Message, "Report error",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        protected override void OnFormClosed(FormClosedEventArgs e)
        {
            if (_report != null) _report.Dispose();
            base.OnFormClosed(e);
        }
    }
}
```

> الـ Viewer فيه أزرار جاهزة: **Print** و **Export** (PDF / Excel / Word). مش محتاج تكتب حاجة لده.

> لو `InventoryReport` مش متعرّفة: اتأكد إن الـ `using InventorySystem.Reports;` موجود، وإن الملف `.rpt` موجود جوه فولدر Reports، وإن الـ Build نجح.

# الوحدة 13: Telerik Grid (اختياري ⭐)

**الفكرة:** نبدّل `DataGridView` بـ `RadGridView` من Telerik. ده بيدّيك بحث وفلترة مدمجين في الجدول نفسه.

## التجهيز

1. نزّل **Telerik UI for WinForms (Free Trial)** وثبّته.
2. افتح المشروع. من الـ Toolbox هتلاقي تاب **Telerik UI for WinForms**. (لو لا: كليك يمين على Toolbox ← Choose Items ← Browse ← DLLs من فولدر Telerik المتثبت.) وتأكد إن الـ References المضافة هي `Telerik.WinControls.dll` و `Telerik.WinControls.UI.dll` و `TelerikCommon.dll` وغيرها اللي بيضيفها VS أوتوماتيك لما تسحب أي Control.
3. افتح `MainForm` في الـ Designer. **امسح** `dgvProducts`، واسحب **RadGridView** مكانه ← (Name) = `gridProducts` ← **Dock = Fill**.

## تعديلات الكود في MainForm.cs

**3) أضف الدوال دي:**

```csharp
private void ConfigureGrid()
{
    gridProducts.ReadOnly = true;
    gridProducts.AllowAddNewRow = false;
    gridProducts.AllowDeleteRow = false;
    gridProducts.AllowEditRow = false;

    gridProducts.EnableFiltering = true;                          // فلترة من أعمدة الجدول
    gridProducts.MasterTemplate.ShowFilteringRow = true;          // صف الفلتر تحت العناوين
    gridProducts.MasterTemplate.AllowSearchRow = true;            // صف البحث فوق الجدول
    gridProducts.EnableGrouping = true;                           // التجميع بالسحب
    gridProducts.MasterTemplate.AutoSizeColumnsMode =
        Telerik.WinControls.UI.GridViewAutoSizeColumnsMode.Fill;
}

private void gridProducts_DataBindingComplete(object sender, GridViewBindingCompleteEventArgs e)
{
    if (gridProducts.Columns.Count == 0) return;

    gridProducts.Columns["ProductId"].HeaderText = "ID";
    gridProducts.Columns["Name"].HeaderText = "Product Name";
    gridProducts.Columns["Quantity"].HeaderText = "Qty in Stock";
    gridProducts.Columns["SupplierName"].HeaderText = "Supplier";
    gridProducts.Columns["CategoryName"].HeaderText = "Category";
    gridProducts.Columns["StockStatus"].HeaderText = "Stock Status";
    gridProducts.Columns["TotalValue"].HeaderText = "Total Value";
    gridProducts.Columns["CategoryId"].IsVisible = false;
    gridProducts.Columns["IsLowStock"].IsVisible = false;

    // تلوين صف Low Stock بالـ Conditional Formatting (مرة واحدة)
    if (gridProducts.Columns["StockStatus"].ConditionalFormattingObjectList.Count == 0)
    {
        var low = new ConditionalFormattingObject("LowStock",
            ConditionTypes.Equal, "Low Stock", "", true);
        low.RowBackColor = System.Drawing.Color.LemonChiffon;
        low.CellBackColor = System.Drawing.Color.LemonChiffon;
        gridProducts.Columns["StockStatus"].ConditionalFormattingObjectList.Add(low);

        var out0 = new ConditionalFormattingObject("OutOfStock",
            ConditionTypes.Equal, "Out of Stock", "", true);
        out0.RowBackColor = System.Drawing.Color.MistyRose;
        out0.CellBackColor = System.Drawing.Color.MistyRose;
        gridProducts.Columns["StockStatus"].ConditionalFormattingObjectList.Add(out0);
    }
}
```

## فلتر جاهز بالكود (مثال)

# ملحق: نقاط تقولها في المناقشة / التسليم

## إزاي حققت كل متطلب؟

- **ADO.NET بدون EF:** `SqlConnection` و `SqlCommand` و `SqlDataAdapter` و `SqlTransaction`، وكلها في طبقة Repositories.
- **أمان الباسورد:** PBKDF2-SHA256 + Salt عشوائي + 600,000 Iteration + مقارنة Constant-time. مفيش باسورد نصي في أي مكان.
- **منع SQL Injection:** كل الـ Queries بـ Parameters.
- **الصلاحيات (RBAC):** اتنين Roles. الواجهة بتخفي الأزرار، **والـ Repository بيرفض** العملية لو المستخدم مش Admin (Defense in Depth).
- **Audit Logging:** بيتسجل في نفس الـ Transaction بتاعة العملية. يعني مفيش تعديل من غير سجل.
- **رسائل الخطأ في Login:** رسالة عامة موحدة + Dummy Hash لو المستخدم مش موجود + حد أقصى 5 محاولات.
- **Constraints في الداتا بيز:** `CHECK` على الكمية والسعر والـ Role، وده بيحمي حتى لو الكود فيه غلط.

## تحسينات ممكن تذكرها كـ "Future Work"

- شاشة لإدارة المستخدمين (إضافة وتعطيل وتغيير الباسورد).
- Lockout مؤقت للحساب بعد محاولات فاشلة (مخزن في الداتا بيز).
- Pagination للجداول الكبيرة.
- Soft Delete بدل الحذف النهائي.
- Unit Tests على الـ Repositories.
